1344. Angle Between Hands of a Clock




class Solution {
public:
    double angleClock(int hour, int minutes) {
        double hourAngle = 30 * hour + 0.5 * minutes;
        double minuteAngle = 6 * minutes;

        double diff = abs(hourAngle - minuteAngle);

        return min(diff, 360 - diff);
    }
};
