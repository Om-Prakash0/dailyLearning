904. Fruit Into Baskets








class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int,int>mp;
        int i = 0;
        int j = 0;
        int n = fruits.size();
        int dist = 0;
        int ans = 0;
        for( i = 0 ; i < n ; i++){
            if(mp[fruits[i]]==0){
                dist++;
            }
            mp[fruits[i]]++;
            if(dist>2){
                while(dist>2){
                    mp[fruits[j]]--;
                    if(mp[fruits[j]]==0){
                        dist--;
                    }
                    j++;
                }
            }
            ans =  max(ans,i-j+1);
        }
        return ans;
    }
};
