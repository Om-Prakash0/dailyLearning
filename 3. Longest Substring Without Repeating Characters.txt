3. Longest Substring Without Repeating Characters




class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int count = 0;
        
        for(int i = 0; i < s.size(); i++) {
            string ans = "";
            
            for(int j = i; j < s.size(); j++) {

                if(ans.find(s[j]) != string::npos) {
                    break;
                }

                ans += s[j];
                count = max(count, (int)ans.size());
            }
        }

        return count;
    }
};
